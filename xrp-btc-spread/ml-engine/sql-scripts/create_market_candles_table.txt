-- Table: public.market_candles

-- DROP TABLE IF EXISTS public.market_candles;

CREATE TABLE IF NOT EXISTS public.market_candles
(
    id bigint NOT NULL DEFAULT nextval('market_candles_id_seq'::regclass),
    exchange text COLLATE pg_catalog."default" NOT NULL,
    symbol text COLLATE pg_catalog."default" NOT NULL,
    "interval" text COLLATE pg_catalog."default" NOT NULL,
    open_time timestamp with time zone NOT NULL,
    close_time timestamp with time zone NOT NULL,
    open numeric(18,8) NOT NULL,
    high numeric(18,8) NOT NULL,
    low numeric(18,8) NOT NULL,
    close numeric(18,8) NOT NULL,
    volume numeric(30,10) NOT NULL,
    quote_volume numeric(30,10),
    trades_count integer,
    taker_buy_base_volume numeric(30,10),
    taker_buy_quote_volume numeric(30,10),
    is_closed boolean DEFAULT true,
    created_at timestamp with time zone DEFAULT now(),
    CONSTRAINT market_candles_pkey PRIMARY KEY (id)
)

TABLESPACE pg_default;

ALTER TABLE IF EXISTS public.market_candles
    OWNER to postgres;
-- Index: idx_market_candles_symbol_interval_time

-- DROP INDEX IF EXISTS public.idx_market_candles_symbol_interval_time;

CREATE INDEX IF NOT EXISTS idx_market_candles_symbol_interval_time
    ON public.market_candles USING btree
    (symbol COLLATE pg_catalog."default" ASC NULLS LAST, "interval" COLLATE pg_catalog."default" ASC NULLS LAST, open_time DESC NULLS FIRST)
    WITH (fillfactor=100, deduplicate_items=True)
    TABLESPACE pg_default;
-- Index: ux_market_candles_uniq

-- DROP INDEX IF EXISTS public.ux_market_candles_uniq;

CREATE UNIQUE INDEX IF NOT EXISTS ux_market_candles_uniq
    ON public.market_candles USING btree
    (exchange COLLATE pg_catalog."default" ASC NULLS LAST, symbol COLLATE pg_catalog."default" ASC NULLS LAST, "interval" COLLATE pg_catalog."default" ASC NULLS LAST, open_time ASC NULLS LAST)
    WITH (fillfactor=100, deduplicate_items=True)
    TABLESPACE pg_default;
