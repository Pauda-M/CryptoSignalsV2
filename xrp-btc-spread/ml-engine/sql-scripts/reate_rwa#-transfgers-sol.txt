CREATE TABLE IF NOT EXISTS rwa_transfers_sol (
    id BIGSERIAL PRIMARY KEY,
    token TEXT NOT NULL,
    wallet_from TEXT,
    wallet_to TEXT,
    amount DOUBLE PRECISION,
    amount_usd DOUBLE PRECISION,
    ts TIMESTAMPTZ NOT NULL,
    signature TEXT,
    raw JSONB,
    UNIQUE(signature, wallet_from, wallet_to)
);
